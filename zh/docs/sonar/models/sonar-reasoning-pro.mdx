    import requests

    url = "https://api.perplexity.ai/v1/sonar"

    payload = {
        "model": "sonar-reasoning-pro",
        "messages": [
            {"role": "user", "content": "Analyze the feasibility of fusion energy becoming a mainstream power source by 2040."}
        ]

        "Content-Type": "application/json"
    }
